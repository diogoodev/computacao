// Função das fases da lua (corrigida)
function createMoonProgressBar(percentage) {
  const totalMoons = 10;
  const progress = percentage / 100;
  const filledMoons = Math.floor(progress * totalMoons);
  const partialProgress = (progress * totalMoons) % 1;
  
  let moonBar = "";
  
  // Luas completas (cheias)
  moonBar += "🌕".repeat(filledMoons);
  
  // Lua parcial baseada no progresso
  if (filledMoons < totalMoons && partialProgress > 0) {
    moonBar += getMoonPhase(partialProgress);
  }
  
  // Luas vazias (novas)
  const remainingMoons = totalMoons - filledMoons - (partialProgress > 0 ? 1 : 0);
  moonBar += "🌑".repeat(Math.max(0, remainingMoons));
  
  return moonBar + `\n${percentage.toFixed(1)}% • ${getMoonMessage(percentage)}`;
}

// Fase da lua baseada no progresso parcial
function getMoonPhase(progress) {
  if (progress <= 0.2) return "🌒"; // Crescente inicial
  if (progress <= 0.4) return "🌓"; // Quarto crescente
  if (progress <= 0.6) return "🌔"; // Gibosa crescente
  if (progress <= 0.8) return "🌕"; // Quase cheia
  return "🌕"; // Cheia
}

// Mensagem motivacional baseada no progresso
function getMoonMessage(percentage) {
  if (percentage >= 95) return "🌕 Lua cheia! Mês quase completo";
  if (percentage >= 80) return "🌔 Gibosa crescente - quase lá!";
  if (percentage >= 60) return "🌓 Quarto crescente - na metade!";
  if (percentage >= 40) return "🌒 Crescente - ganhando momentum";
  if (percentage >= 20) return "🌒 Lua nova crescendo";
  return "🌑 Início do ciclo lunar";
}

function getLunarPhase() {
  const lunarCycle = 29.5; // dias
  const knownNewMoon = moment('2024-01-11'); // referência
  const daysSince = today.diff(knownNewMoon, 'days');
  const phase = (daysSince % lunarCycle) / lunarCycle;
  
  if (phase < 0.0625 || phase >= 0.9375) return "🌑 Nova";
  if (phase < 0.1875) return "🌒 Crescente";
  if (phase < 0.3125) return "🌓 Quarto Crescente";
  if (phase < 0.4375) return "🌔 Gibosa Crescente";
  if (phase < 0.5625) return "🌕 Cheia";
  if (phase < 0.6875) return "🌖 Gibosa Minguante";
  if (phase < 0.8125) return "🌗 Quarto Minguante";
  return "🌘 Minguante";
}
